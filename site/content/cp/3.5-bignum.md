# 3.5 High-Precision Arithmetic 高精度运算

When numbers go past `long long` (about $9.2 \times 10^{18}$), store them as **arrays of digits** and do the arithmetic by hand, the same way as column arithmetic on paper.

Store digits **little-endian**: `d[0]` is the ones digit. Then position `i` means $10^i$, and a carry always moves to `i+1`.

## The four operations

| Operation | Idea | Cost |
|---|---|---|
| Addition | `s = x + y + carry; c[i] = s % 10; carry = s / 10;` from the lowest digit up | $O(n)$ |
| Subtraction | Subtract digit by digit; if $s < 0$, add 10 and **borrow** 1 from the next digit. Compare first so the bigger number is on top. | $O(n)$ |
| Multiplication | Add every product `a[i]*b[j]` into bucket `c[i+j]` with **no** carrying, then do **one** carry pass at the end | $O(nm)$ |
| Division by a small int | Go from the **highest** digit down: `cur = r*10 + a[i]; c[i] = cur / m; r = cur % m;` | $O(n)$ |

After subtraction, multiplication and division, call `trim()` to remove leading zeros (but keep one digit for 0).

## Full code

```cpp
#include<bits/stdc++.h>
using namespace std;

// Digits are stored little-endian: d[0] is the ones digit.
struct Big {
    vector<int> d;
    Big(string s = "0") {
        for (int i = (int)s.size() - 1; i >= 0; i--) d.push_back(s[i] - '0');
        trim();
    }
    void trim() {                       // drop leading zeros, keep at least one digit
        while (d.size() > 1 && d.back() == 0) d.pop_back();
    }
    string str() const {
        string s;
        for (int i = (int)d.size() - 1; i >= 0; i--) s += char('0' + d[i]);
        return s;
    }
};

// |a| < |b| ?  (both already trimmed)
bool lessThan(const Big &a, const Big &b) {
    if (a.d.size() != b.d.size()) return a.d.size() < b.d.size();
    for (int i = (int)a.d.size() - 1; i >= 0; i--)
        if (a.d[i] != b.d[i]) return a.d[i] < b.d[i];
    return false;
}

Big add(const Big &a, const Big &b) {
    int n = max(a.d.size(), b.d.size());
    Big c; c.d.assign(n + 1, 0);
    int carry = 0;
    for (int i = 0; i < n; i++) {
        int x = 0;
        if (i < a.d.size()) x = a.d[i];

        int y = 0;
        if (i < b.d.size()) y = b.d[i];

        int s = x + y + carry;
        c.d[i] = s % 10;
        carry = s / 10;
    }
    c.d[n] = carry;
    c.trim();
    return c;
}

// requires a >= b
Big sub(const Big &a, const Big &b) {
    Big c; c.d.assign(a.d.size(), 0);
    int borrow = 0;
    for (int i = 0; i < a.d.size(); i++) {
        int x = a.d[i] - borrow;

        int y = 0;
        if (i < b.d.size()) y = b.d[i];

        int s = x - y;
        if (s < 0) { s += 10; borrow = 1; } else borrow = 0;
        c.d[i] = s;
    }
    c.trim();
    return c;
}

Big mul(const Big &a, const Big &b) {
    Big c; c.d.assign(a.d.size() + b.d.size(), 0);
    for (int i = 0; i < a.d.size(); i++)
      for (int j = 0; j < b.d.size(); j++)
        c.d[i+j] += a.d[i] * b.d[j];

    for (int i = 0; i+1 < c.d.size(); i++) {
      c.d[i+1] += c.d[i] / 10;
      c.d[i] %= 10;
    }
    c.trim();
    return c;
}

// big ÷ small int m; the remainder is returned through r
Big divInt(const Big &a, int m, long long &r) {
    Big c; c.d.assign(a.d.size(), 0);
    r = 0;
    for (int i = a.d.size()-1; i >= 0; i--) {
        long long cur = r * 10 + a.d[i];
        c.d[i] = cur / m;
        r = cur % m;
    }
    c.trim();
    return c;
}

int main() {
    string sa, sb;
    int m;
    cin >> sa >> sb >> m;
    Big a(sa), b(sb);

    cout << "a + b = " << add(a, b).str() << '\n';
    if (lessThan(a, b)) cout << "a - b = -" << sub(b, a).str() << '\n';
    else                cout << "a - b = " << sub(a, b).str() << '\n';
    cout << "a * b = " << mul(a, b).str() << '\n';
    long long r;
    Big q = divInt(a, m, r);
    cout << "a / " << m << " = " << q.str() << " remainder " << r << '\n';
    return 0;
}
```
